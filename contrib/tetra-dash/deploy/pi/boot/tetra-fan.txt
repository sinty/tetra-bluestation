# Вентилятор платы: крутится всегда, с нагревом ускоряется.
#
# Ставится в /boot/firmware/tetra-fan.txt, подключается строкой
#
#     include tetra-fan.txt
#
# в /boot/firmware/config.txt.
#
# Плата Waveshare CM4-DUAL-ETH-MINI: управление вентилятором на GPIO19,
# датчик оборотов на GPIO17. Микросхемы управления на плате нет.
#
# Почему не одной строкой: длинная строка dtoverlay=… в config.txt обрезается,
# и часть параметров молча теряется (проверено 18.09.2026 — порог 65 °C
# не применился). Поэтому оверлей задаётся коротко, а параметры идут
# отдельными строками dtparam=: они применяются к последнему оверлею.
#
# Почему пороги, а не «минимальные обороты»: нулевая ступень оверлея всегда
# равна нулю, параметрами её не поднять — на ней вентилятор стоит. Поэтому
# первый порог сдвинут к 0 °C с нулевым гистерезисом: ступень «стоять»
# недостижима, и самая низкая ступень — это минимальные обороты.
#
# Уровни PWM и обороты кулера Raspberry Pi 5, измерено 18.09.2026:
#
#     255 → 6970 об/мин       80 → 2470 об/мин
#     120 → 3510 об/мин       50 → 1350 об/мин
#     100 → 3000 об/мин       35 →  750 об/мин   (почти не слышно)
#
# Ниже 25 кулер срывается: на 20 остаётся 190 об/мин. С остановки трогается
# начиная с 30. Поэтому минимум — 35, с запасом.
#
# Итоговая раскладка:
#
#     всегда      35  (~750 об/мин)   минимальный продув
#     от 55 °C   114  (~3500 об/мин)
#     от 65 °C   152
#     от 75 °C   255  (~7000 об/мин)  с 80 °C процессор снижает частоту
#
# Чтобы вернуть штатное поведение (ниже 50 °C вентилятор стоит), достаточно
# убрать строку include из config.txt.

dtoverlay=pwm-gpio-fan
dtparam=fan_gpio=19
dtparam=fan_temp0=0
dtparam=fan_temp0_hyst=0
dtparam=fan_temp0_speed=35
dtparam=fan_temp1=55000
dtparam=fan_temp1_speed=114
dtparam=fan_temp2=65000
dtparam=fan_temp2_speed=152
dtparam=fan_temp3=75000
dtparam=fan_temp3_speed=255
